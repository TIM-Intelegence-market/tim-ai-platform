"""Fetch public Binance USD-M monthly fundingRate zips (data.binance.vision) for the symbols in the S32 training universe, 2020-01..2022-12 only."""
import concurrent.futures as cf, json, sys, time, urllib.request
from pathlib import Path
CDN = "https://data.binance.vision/data/futures/um/monthly/fundingRate"
uni = json.load(open(sys.argv[1])); root = Path(sys.argv[2])
syms = sorted({s for v in uni.values() for s in v})
months = [f"{y}-{m:02d}" for y in range(2020, 2023) for m in range(1, 13)]
jobs = [(s, ym) for s in syms for ym in months]
def get(j):
    s, ym = j; dest = root / s / f"{s}-fundingRate-{ym}.zip"
    if dest.exists(): return "have"
    dest.parent.mkdir(parents=True, exist_ok=True)
    for a in range(3):
        try:
            dest.write_bytes(urllib.request.urlopen(f"{CDN}/{s}/{s}-fundingRate-{ym}.zip", timeout=60).read()); return "ok"
        except Exception as e:
            if getattr(e, "code", None) == 404: return "404"
            time.sleep(2 * (a + 1))
    return "fail"
res = {}
with cf.ThreadPoolExecutor(4) as ex:
    for st in ex.map(get, jobs): res[st] = res.get(st, 0) + 1
print(len(syms), "symbols", len(jobs), "files", res)
